\usepackage[margin=0.55in]{geometry}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{parskip}
\usepackage{hyperref}

% Disable paragraph indentation
\setlength{\parindent}{0pt}

% Compact paragraphs
\setlength{\parskip}{2pt}

% Compact bullet lists
\setlist[itemize]{
  leftmargin=0.20in,
  itemsep=1pt,
  topsep=2pt,
  parsep=0pt,
  partopsep=0pt
}

% Main section headings
\titleformat{\section}
  {\large\bfseries}
  {}
  {0pt}
  {}
  [\titlerule]

\titlespacing*{\section}
  {0pt}
  {8pt}
  {4pt}

% Job/project headings
\titleformat{\subsection}
  {\normalsize\bfseries}
  {}
  {0pt}
  {}

\titlespacing*{\subsection}
  {0pt}
  {5pt}
  {1pt}

% Prevent ugly automatic word splitting
\hyphenpenalty=10000
\exhyphenpenalty=10000

% Links
\hypersetup{
  colorlinks=true,
  urlcolor=black,
  linkcolor=black
}

% Slightly compact line spacing
\linespread{0.95}
